import 'package:flutter/foundation.dart';

import '../playback/radar_playback_snapshot.dart';
import 'radar_playback_source.dart';

/// UI-Lebenszykluscontroller für die Radar-Playback-Integration.
///
/// Eine Flutter-Seite kann diesen Controller in `initState` initialisieren,
/// über einen `ListenableBuilder` oder `AnimatedBuilder` beobachten und in
/// `dispose` wieder freigeben.
///
/// Die bestehende Kartenansicht muss dadurch weder Provider- noch
/// Render-Details der Weather Engine kennen.
class RadarPlaybackViewController<TOutput> extends ChangeNotifier {
  RadarPlaybackViewController({
    required this._source,
    this.disposeSource = false,
  }) {
    _source.addListener(_handleSourceChanged);
  }

  final RadarPlaybackSource<TOutput> _source;

  /// Gibt an, ob der Controller beim eigenen Dispose auch die angebundene
  /// Radarquelle freigeben soll.
  ///
  /// Bleibt der Besitz bei einem übergeordneten Controller oder Service,
  /// muss dieser Wert `false` bleiben.
  final bool disposeSource;

  bool _initialized = false;
  bool _initializing = false;
  bool _disposed = false;

  Future<void>? _initializationOperation;

  RadarPlaybackSnapshot<TOutput> get snapshot => _source.snapshot;

  bool get isInitialized => _initialized;

  bool get isInitializing => _initializing;

  bool get isDisposed => _disposed;

  RadarPlaybackSource<TOutput> get source => _source;

  /// Initialisiert die Radarquelle höchstens einmal.
  ///
  /// Gleichzeitige Aufrufe erhalten dieselbe laufende Future. Dadurch werden
  /// doppelte Timeline-Anfragen beim Aufbau eines Widgets verhindert.
  Future<void> initialize() {
    _ensureActive();

    if (_initialized) {
      return Future<void>.value();
    }

    final runningOperation = _initializationOperation;

    if (runningOperation != null) {
      return runningOperation;
    }

    final operation = _runInitialization();
    _initializationOperation = operation;

    return operation;
  }

  Future<void> _runInitialization() async {
    _initializing = true;
    notifyListeners();

    try {
      await _source.initialize();
      _initialized = true;
    } finally {
      _initializing = false;
      _initializationOperation = null;

      if (!_disposed) {
        notifyListeners();
      }
    }
  }

  /// Fordert eine neue Radar-Timeline über die angebundene Quelle an.
  Future<void> refresh() {
    _ensureActive();
    return _source.refresh();
  }

  void _handleSourceChanged() {
    if (_disposed) {
      return;
    }

    notifyListeners();
  }

  void _ensureActive() {
    if (_disposed) {
      throw StateError(
        'RadarPlaybackViewController wurde bereits freigegeben.',
      );
    }
  }

  @override
  void dispose() {
    if (_disposed) {
      return;
    }

    _disposed = true;
    _source.removeListener(_handleSourceChanged);

    if (disposeSource) {
      _source.dispose();
    }

    super.dispose();
  }
}

import 'package:flutter/foundation.dart';

/// Produktive Verbindung zwischen dem bestehenden
/// [RainViewerRadarService] und der allgemeinen Radar-Playback-Integration.
///
/// Die Source übernimmt:
///
/// - das Laden der RainViewer-Metadaten,
/// - die Abbildung auf providerneutrale [WeatherFrame]-Objekte,
/// - die Auswahl des aktuellsten RainViewer-Radarframes,
/// - Refresh- und Fehlerzustände,
/// - die Benachrichtigung angebundener Flutter-Controller.
///
/// Animation, Kartenanzeige und Layer-Überblendung verbleiben weiterhin
/// vollständig in der Radaroberfläche.
class RainViewerRadarPlaybackSource extends ChangeNotifier
    implements RadarPlaybackSource<RainViewerRadarFrame> {
  RainViewerRadarPlaybackSource({required this.radarService});

  static const String providerId = 'rainviewer';

  final RainViewerRadarService radarService;

  RadarPlaybackSnapshot<RainViewerRadarFrame> _snapshot =
      const RadarPlaybackSnapshot<RainViewerRadarFrame>(
        status: RadarPlaybackStatus.idle,
      );

  bool _initialized = false;
  bool _loading = false;
  bool _disposed = false;

  Future<void>? _pendingOperation;

  @override
  RadarPlaybackSnapshot<RainViewerRadarFrame> get snapshot => _snapshot;

  bool get isInitialized => _initialized;

  bool get isLoading => _loading;

  bool get isDisposed => _disposed;
  @override
  Future<void> initialize() {
    _ensureActive();

    if (_initialized) {
      return Future<void>.value();
    }

    return _load(forceRefresh: false);
  }

  @override
  Future<void> refresh() {
    _ensureActive();

    return _load(forceRefresh: true);
  }

  Future<void> _load({required bool forceRefresh}) {
    final runningOperation = _pendingOperation;

    if (runningOperation != null) {
      return runningOperation;
    }

    final operation = _performLoad(forceRefresh: forceRefresh);
    _pendingOperation = operation;

    return operation.whenComplete(() {
      if (identical(_pendingOperation, operation)) {
        _pendingOperation = null;
      }
    });
  }

  Future<void> _performLoad({required bool forceRefresh}) async {
    _loading = true;

    _updateSnapshot(
      _snapshot.copyWith(
        status: RadarPlaybackStatus.loadingTimeline,
        clearError: true,
      ),
    );

    try {
      final metadata = await radarService.fetchMetadata(
        forceRefresh: forceRefresh,
      );

      if (_disposed) {
        return;
      }

      final rainViewerFrames = metadata.frames;

      if (rainViewerFrames.isEmpty) {
        _initialized = true;

        _updateSnapshot(
          const RadarPlaybackSnapshot<RainViewerRadarFrame>(
            status: RadarPlaybackStatus.ready,
          ),
        );

        return;
      }

      final weatherFrames = rainViewerFrames
          .map(
            (frame) => _toWeatherFrame(
              frame: frame,
              generatedAt: metadata.generatedAt,
            ),
          )
          .toList(growable: false);

      final latestIndex = rainViewerFrames.length - 1;
      final latestFrame = rainViewerFrames[latestIndex];

      _initialized = true;

      _updateSnapshot(
        RadarPlaybackSnapshot<RainViewerRadarFrame>(
          status: RadarPlaybackStatus.ready,
          frames: List<WeatherFrame>.unmodifiable(weatherFrames),
          currentIndex: latestIndex,
          currentOutput: latestFrame,
          isPreloading: false,
          preloadedFrameCount: 0,
        ),
      );
    } catch (error, stackTrace) {
      if (_disposed) {
        return;
      }

      _updateSnapshot(
        _snapshot.copyWith(
          status: RadarPlaybackStatus.failed,
          error: error,
          stackTrace: stackTrace,
          isPreloading: false,
        ),
      );

      rethrow;
    } finally {
      _loading = false;
    }
  }

  WeatherFrame _toWeatherFrame({
    required RainViewerRadarFrame frame,
    required DateTime generatedAt,
  }) {
    return WeatherFrame(
      id: 'rainviewer-radar-${frame.time.millisecondsSinceEpoch}',
      layerType: WeatherLayerType.radar,
      validTime: frame.time,
      generatedAt: generatedAt,
      providerId: providerId,
      state: WeatherFrameState.ready,
      sourceReference: frame.path,
    );
  }

  void _updateSnapshot(
    RadarPlaybackSnapshot<RainViewerRadarFrame> nextSnapshot,
  ) {
    if (_disposed) {
      return;
    }

    _snapshot = nextSnapshot;
    notifyListeners();
  }

  void _ensureActive() {
    if (_disposed) {
      throw StateError(
        'RainViewerRadarPlaybackSource wurde bereits freigegeben.',
      );
    }
  }

  @override
  void dispose() {
    if (_disposed) {
      return;
    }

    _disposed = true;
    _loading = false;

    _snapshot = _snapshot.copyWith(status: RadarPlaybackStatus.disposed);

    super.dispose();
  }
}

    _animationTimer?.cancel();

    _animationTimer = Timer(const Duration(milliseconds: 1900), () {
      if (!mounted || !_isAnimating) {
        return;
      }

      final metadata = _metadata;

      if (metadata == null || metadata.frames.length < 2) {
        _stopAnimation();
        return;
      }

      final nextIndex = (_selectedFrameIndex + 1) % metadata.frames.length;

      setState(() {
        if (_showPrimaryRadarLayer) {
          _secondaryRadarFrameIndex = nextIndex;
        } else {
          _primaryRadarFrameIndex = nextIndex;
        }

        setState(() {
          _selectedFrameIndex = nextIndex;
          _showPrimaryRadarLayer = !_showPrimaryRadarLayer;
        });

        _scheduleNextRadarFrame();

    });
